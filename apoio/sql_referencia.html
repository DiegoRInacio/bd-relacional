Referência de SQL

Cola de sintaxe do PostgreSQL para consultar durante exercícios e provas. Todos os exemplos usam o esquema abaixo. Copie e adapte.

Esquema de exemplo

CREATE TABLE cliente (
    id      SERIAL       PRIMARY KEY,
    nome    VARCHAR(100) NOT NULL,
    cidade  VARCHAR(60)
);

CREATE TABLE produto (
    id     SERIAL       PRIMARY KEY,
    nome   VARCHAR(100) NOT NULL,
    preco  NUMERIC(10,2) NOT NULL CHECK (preco > 0)
);

CREATE TABLE pedido (
    id          SERIAL      PRIMARY KEY,
    id_cliente  INT         NOT NULL REFERENCES cliente(id),
    data_pedido DATE        NOT NULL DEFAULT CURRENT_DATE,
    status      VARCHAR(15) NOT NULL DEFAULT 'aberto'
);

CREATE TABLE item_pedido (
    id_pedido  INT NOT NULL REFERENCES pedido(id),
    id_produto INT NOT NULL REFERENCES produto(id),
    quantidade INT NOT NULL CHECK (quantidade > 0),
    PRIMARY KEY (id_pedido, id_produto)
);

DDL — definir a estrutura

CREATE / DROP

CREATE TABLE nome ( coluna TIPO [constraints], ... );
DROP TABLE nome;
DROP TABLE IF EXISTS nome;           -- não dá erro se não existir
DROP TABLE nome CASCADE;             -- derruba também o que depende dela
TRUNCATE nome;                       -- esvazia (mais rápido que DELETE)

DML — mexer nos dados

INSERT

-- sempre liste as colunas
INSERT INTO cliente (nome, cidade) VALUES ('Ana', 'SP');

-- várias linhas de uma vez
INSERT INTO produto (nome, preco) VALUES
  ('Notebook', 3000), ('Mouse', 80), ('Teclado', 150);

-- deixar o SERIAL / DEFAULT agir: só não informe a coluna
INSERT INTO pedido (id_cliente) VALUES (1);

-- a partir de uma consulta
INSERT INTO cliente_vip (nome)
SELECT nome FROM cliente WHERE cidade = 'SP';

UPDATE

UPDATE cliente SET cidade = 'RJ' WHERE id = 1;

-- sem WHERE muda TODAS as linhas ⚠
UPDATE pedido SET status = 'fechado' WHERE data_pedido < '2024-01-01';

-- usando outra tabela
UPDATE produto p SET preco = preco * 1.10
FROM   reajuste r WHERE r.id_produto = p.id;

DELETE

DELETE FROM pedido WHERE status = 'cancelado';

-- ordem por causa das FKs: filho antes do pai
DELETE FROM item_pedido WHERE id_pedido = 5;
DELETE FROM pedido      WHERE id = 5;

-- sem WHERE apaga a tabela inteira ⚠

DQL — consultar

SELECT nome, cidade          -- colunas (ou *)
FROM   cliente
WHERE  cidade = 'SP'         -- filtro de linhas
ORDER BY nome ASC            -- ASC é o padrão; DESC inverte
LIMIT  10;                   -- só as primeiras N

SELECT DISTINCT cidade FROM cliente;   -- valores sem repetição

JOIN — juntar tabelas

-- INNER JOIN: só as linhas que casam nas duas pontas
SELECT c.nome, p.id, p.status
FROM   cliente c
JOIN   pedido  p ON p.id_cliente = c.id;

-- LEFT JOIN: todos os clientes, mesmo os sem pedido (colunas de pedido = NULL)
SELECT c.nome, p.id
FROM   cliente c
LEFT JOIN pedido p ON p.id_cliente = c.id;

-- vários JOINs encadeados
SELECT c.nome, pr.nome AS produto, i.quantidade
FROM   pedido p
JOIN   cliente     c  ON c.id = p.id_cliente
JOIN   item_pedido i  ON i.id_pedido = p.id
JOIN   produto     pr ON pr.id = i.id_produto;
Apelidos (alias): cliente c deixa escrever c.nome. Em JOIN, sempre qualifique a coluna (c.id, p.id) para evitar ambiguidade.

Agregação

FunçãoFaz
COUNT(*)conta linhas · COUNT(col) ignora NULL
SUM(col)soma
AVG(col)média
MIN(col) / MAX(col)menor / maior
-- resumo geral
SELECT COUNT(*), AVG(preco) FROM produto;

-- resumo por categoria: GROUP BY
SELECT p.status,
       COUNT(*)                       AS qtd,
       SUM(i.quantidade * pr.preco)   AS total
FROM   pedido p
JOIN   item_pedido i ON i.id_pedido = p.id
JOIN   produto     pr ON pr.id = i.id_produto
GROUP BY p.status
HAVING SUM(i.quantidade * pr.preco) > 1000   -- filtra DEPOIS de agrupar
ORDER BY total DESC;
Regra: toda coluna do SELECT que não é agregação precisa estar no GROUP BY.   WHERE filtra antes de agrupar, HAVING filtra os grupos.

Transações

BEGIN;
  UPDATE conta SET saldo = saldo - 100 WHERE id = 1;
  UPDATE conta SET saldo = saldo + 100 WHERE id = 2;
COMMIT;        -- confirma tudo
-- ou:
ROLLBACK;      -- desfaz tudo
Ótimo para testar um UPDATE/DELETE arriscado: roda dentro de BEGIN, confere com um SELECT, e só então COMMIT (ou ROLLBACK para cancelar).
